# Разбор кода:
movl $0, -4(%rbp) — инициализирует аккумулятор значением 0. Назовем её total.
movl $1, -8(%rbp) — инициализирует счетчик цикла значением 1. Назовем его i.
jmp L2 — безусловный переход к метке L2 для проверки условия.
Тело цикла (метка L3):
movl -8(%rbp), %eax — копирует значение счетчика i в регистр %eax.
addl %eax, -4(%rbp) — прибавляет значение %eax (текущее i) к аккумулятору total.
addl $1, -8(%rbp) — увеличивает счетчик i на 1 (i++).
Проверка условия (метка L2):
cmpl $10, -8(%rbp) — сравнивает текущее значение счетчика i с числом 10.
jle L3 — если i меньше или равно 10 (jump if less or equal), возвращается к телу цикла L3.
Выход из цикла:
movl -4(%rbp), %eax — помещает финальную сумму из аккумулятора в регистр %eax.

# Код на python:
def calculate_sum():
total = 0  
i = 1      
while i <= 10:
total += i
i += 1
return total
